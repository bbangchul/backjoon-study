# 리스트 정리 노트

이 폴더의 리스트는 삽입할 때 오름차순을 유지합니다.

| 함수 | 역할 |
| --- | --- |
| `GetSize()` | 원소 개수 반환 |
| `GetAt(index)` | 해당 인덱스의 값 반환 |
| `Index(value)` | 해당 값의 인덱스 반환, 없으면 -1 |
| `Find(value)` | 값이 있는지 확인 |
| `Insert(value)` | 정렬된 위치에 삽입 |
| `Delete(value)` | 같은 값의 첫 원소 하나 삭제 |

### 배열 리스트

삽입할 자리 뒤의 값을 **오른쪽으로**, 삭제한 자리 뒤의 값을 **왼쪽으로** 옮깁니다.

```text
Insert(3)
[2][5][7][?]  ->  [2][5][5][7]  ->  [2][3][5][7]
                 뒤에서부터 이동     빈자리에 3 저장

Delete(3)
[2][3][5][7]  ->  [2][5][7] | [7]
                 유효 값     미사용 칸에 남은 값
```

삭제 후 마지막 칸에 값이 남아 있어도 현재 개수 밖이면 유효 데이터가 아닙니다.

### 단일 연결 리스트

노드를 따라 이동합니다. 배열처럼 연속한 메모리에 저장할 필요가 없습니다.

```text
lst_ --> [2|next] --> [5|next] --> [7|0]

Insert(3)
lst_ --> [2|next] --> [3|next] --> [5|next] --> [7|0]

Delete(3)
lst_ --> [2|next] --------------> [5|next] --> [7|0]
                     [3]은 연결에서 제외한 뒤 delete
```

삽입할 때 기존 다음 노드의 주소를 보존하고 연결을 바꿉니다.
삭제할 때는 노드를 건너뛰도록 연결한 뒤 메모리를 해제합니다.

### 이중 연결 리스트

각 노드가 이전 노드와 다음 노드의 주소를 모두 저장합니다.

```text
         head_                         tail_
           |                             |
           v                             v
0 <-- [prev|2|next] <--> [prev|3|next] <--> [prev|5|next] --> 0
```

`Print()`는 head부터 next 방향, `PrintReverse()`는 tail부터 prev 방향으로 이동합니다.
삽입·삭제 시 양쪽 연결을 모두 바꿔야 합니다. 첫 노드와 마지막 노드는 별도로 처리합니다.

## 전체 코드

각 코드 블록은 **별도의 완전한 프로그램**입니다. 서로 합치지 않고 하나씩 컴파일합니다.
같은 내용을 `코드/` 폴더의 `.cpp` 파일에도 저장했습니다. 원래 수업 폴더는 유지했습니다.
노트용 코드는 중복 클래스 정의를 제거하고, 누락된 Find 구현·인덱스 검사·메모리 해제를 필요한 곳에 보완했습니다.

이 폴더에서 다음처럼 실행합니다(C++11 이상).

### 배열 리스트

정렬된 위치에 삽입하며, 삭제 뒤의 값을 왼쪽으로 옮깁니다.

[코드 파일](코드/list_array.cpp)

```sh
c++ -std=c++11 코드/list_array.cpp -o /tmp/list_array
/tmp/list_array
```

```cpp

// [알고리즘 C.3] list.cpp
#include <iostream>
using namespace std;
class list
{
public:
    list(int lsize = 100);
    ~list() { delete[] arr; }
    list(const list &) = delete;
    list &operator=(const list &) = delete;
    int GetSize();
    int GetAt(int index);
    int Index(int key); // index Rank(key item);
    void Insert(int item);
    void Delete(int key);
    int Find(int value);

private:
    int LMAX;
    int *arr;
    int size;
};
// 여기까지 list.h

list::list(int lsize)
{
    arr = new int[lsize];
    LMAX = lsize;
    size = 0;
}

int list::GetSize()
{
    return size;
}

int list::GetAt(int position)
{
    if (position < 0 || size <= position)
        return -1;
    return arr[position];
}

int list::Index(int value)
{
    for (int i = 0; i < size; i++)
        if (value == arr[i])
            return i;
    return -1;
}

int list::Find(int value)
{
    return Index(value) != -1;
}

void list::Insert(int item)
{
    int i;
    if (size == LMAX)
        return;
    for (i = size; 0 < i; i--)
    {
        if (item < arr[i - 1])
            arr[i] = arr[i - 1];
        else
            break;
    }
    arr[i] = item;
    size++;
}

void list::Delete(int value)
{
    int i;
    for (i = 0; i < size; i++)
        if (value == arr[i])
            break;
    if (i == size)
        return;
    size--;
    for (; i < size; i++)
        arr[i] = arr[i + 1];
}

int main()
{
    list lst;
    lst.Insert(5); lst.Insert(2); lst.Insert(3);
    cout << "GetSize: " << lst.GetSize() << endl;
    cout << "GetAt(1): " << lst.GetAt(1) << endl;
    cout << "Index(5): " << lst.Index(5) << endl;
    cout << "Find(3): " << lst.Find(3) << endl;
    lst.Delete(3);
    cout << "Delete(3) 후: ";
    for (int i = 0; i < lst.GetSize(); i++) cout << lst.GetAt(i) << " ";
    cout << endl;
}
```

```cpp
// [알고리즘 C.5] ptrlist.cpp
class node
{
public:
    node(int d, node *n = 0)
    {
        nxt_ = n;
        data_ = d;
    }
    // private: 코딩상의 편의를 위하여 모두 public으로 한다.
    node *nxt_;
    int data_;
};

class list
{
public:
    list();
    ~list();
    list(const list &) = delete;
    list &operator=(const list &) = delete;
    int GetSize();
    int GetAt(int index);
    int Index(int key);
    void Insert(int item);
    void Delete(int key);
    int Find(int key);

private:
    node *lst_;
};
// 여기까지 ptrlist.h

list::list()
{
    lst_ = 0;
    // size_= 0;
}

int list::GetSize()
{
    node *ptr = lst_;
    int count = 0;
    while (ptr != 0)
    {
        ptr = ptr->nxt_;
        count++;
    }
    return count;
}

int list::GetAt(int index)
{
    if (index < 0) return -1;
    node *ptr = lst_;
    for (int i = 0; i < index; i++)
    {
        if (ptr == 0)
            return -1;
        ptr = ptr->nxt_;
    }
    return ptr == 0 ? -1 : ptr->data_;
}

int list::Index(int key)
{
    node *ptr = lst_;
    int idx = 0;
    while (ptr != 0)
    {
        if (ptr->data_ == key)
            return idx;
        ptr = ptr->nxt_;
        idx++;
    }
    return -1;
}

void list::Insert(int item)
{
    if (lst_ == 0)
    {
        lst_ = new node(item, 0);
        return;
    }
    if (item < lst_->data_)
    {
        lst_ = new node(item, lst_);
        return;
    }
    node *ptr = lst_;
    while (ptr->nxt_ != 0 && ptr->nxt_->data_ < item)
    {
        ptr = ptr->nxt_;
    }
    ptr->nxt_ = new node(item, ptr->nxt_);
}



void list::Delete(int key)
{
    node *ptr = lst_;
    if (lst_ == 0)
        return;
    if (lst_->data_ == key)
    {
        lst_ = lst_->nxt_;
        delete ptr;
        return;
    }
    while (ptr->nxt_ != 0 && ptr->nxt_->data_ != key)
    {
        ptr = ptr->nxt_;
    }
    if (ptr->nxt_ == 0)
        return;
    node *dptr = ptr->nxt_;
    ptr->nxt_ = ptr->nxt_->nxt_;
    delete dptr;
}



#include <iostream>
using namespace std;

list::~list()
{
    while (lst_ != 0)
    {
        node *ptr = lst_;
        lst_ = lst_->nxt_;
        delete ptr;
    }
}

int list::Find(int key)
{
    return Index(key) != -1;
}

int main()
{
    list lst;
    lst.Insert(5); lst.Insert(2); lst.Insert(3);
    cout << "GetSize: " << lst.GetSize() << endl;
    cout << "GetAt(1): " << lst.GetAt(1) << endl;
    cout << "Index(5): " << lst.Index(5) << endl;
    cout << "Find(3): " << lst.Find(3) << endl;
    lst.Delete(3);
    cout << "Delete(3) 후: ";
    for (int i = 0; i < lst.GetSize(); i++) cout << lst.GetAt(i) << " ";
    cout << endl;
}
```

### 이중 연결 리스트

prev와 next를 모두 관리합니다. 같은 첫 값의 삽입도 처리하도록 보완했습니다.

[코드 파일](코드/list_double.cpp)

```sh
c++ -std=c++11 코드/list_double.cpp -o /tmp/list_double
/tmp/list_double
```

```cpp


#include <iostream>

using namespace std;

class dlnode
{
public:

    dlnode(int d, dlnode *p = 0, dlnode *n = 0)
    {
        prev_ = p;
        next_ = n;
        data_ = d;
    }
    dlnode *prev_;
    dlnode *next_;
    int data_;
};

class dllist
{
public:
    dllist();
    ~dllist();
    dllist(const dllist &) = delete;
    dllist &operator=(const dllist &) = delete;

    void Insert(int item);
    void Delete(int value);

    int Find(int key);
    void Print();
    void PrintReverse();

private:

    dlnode *head_;
    dlnode *tail_;
};

dllist::dllist()
{
    head_ = 0;
    tail_ = 0;
}

void dllist::Insert(int item)
{

    if (head_ == 0)
    {
        head_ = new dlnode(item);
        tail_ = head_;
        return;
    }

    if (item <= head_->data_)
    {
        dlnode *ptr = new dlnode(item, 0, head_);
        head_->prev_ = ptr;
        head_ = ptr;
        return;
    }

    if (tail_->data_ < item)
    {
        dlnode *ptr = new dlnode(item, tail_, 0);
        tail_->next_ = ptr;
        tail_ = ptr;
        return;
    }

    dlnode *ptr = head_;
    while (ptr->data_ < item)
        ptr = ptr->next_;

    dlnode *nptr = new dlnode(item, ptr->prev_, ptr);
    ptr->prev_->next_ = nptr;
    ptr->prev_ = nptr;
}

void dllist::Delete(int value)
{

    dlnode *ptr = head_;
    while (ptr != 0 && ptr->data_ != value)
        ptr = ptr->next_;
    if (ptr == 0)
        return;

    if (ptr->prev_ == 0)
        head_ = ptr->next_;
    else
        ptr->prev_->next_ = ptr->next_;

    if (ptr->next_ == 0)
        tail_ = ptr->prev_;
    else
        ptr->next_->prev_ = ptr->prev_;

    delete ptr;
}

void dllist::Print()
{
    for (dlnode *ptr = head_; ptr != 0; ptr = ptr->next_)
        cout << ptr->data_ << " ";
    cout << endl;
}

void dllist::PrintReverse()
{
    for (dlnode *ptr = tail_; ptr != 0; ptr = ptr->prev_)
        cout << ptr->data_ << " ";
    cout << endl;
}

dllist::~dllist()
{
    while (head_ != 0)
    {
        dlnode *ptr = head_;
        head_ = head_->next_;
        delete ptr;
    }
}

int dllist::Find(int key)
{
    for (dlnode *ptr = head_; ptr != 0; ptr = ptr->next_)
        if (ptr->data_ == key) return 1;
    return 0;
}

int main()
{
    dllist lst;
    lst.Insert(5); lst.Insert(2); lst.Insert(3); lst.Insert(2);
    cout << "Print: "; lst.Print();
    cout << "PrintReverse: "; lst.PrintReverse();
    cout << "Find(3): " << lst.Find(3) << endl;
    lst.Delete(3);
    cout << "Delete(3) 후: "; lst.Print();
}
```

## 시간복잡도

n은 현재 원소 개수이며, 메모리 할당 비용은 별도입니다.

| 구현 | 삽입 | 값으로 삭제 | 인덱스 조회 |
| --- | --- | --- | --- |
| 배열 | O(n) | O(n) | O(1) |
| 단일 연결 | O(n) | O(n) | O(n) |
| 이중 연결 | 최악 O(n) | O(n), 찾은 뒤 연결 변경 O(1) | 현재 예제에는 없음 |
