# 큐 정리 노트

FIFO(First In, First Out): 먼저 넣은 값을 먼저 꺼냅니다.

```text
DeleteQ() <-- [2][5][3] <-- InsertQ(7)
              앞   뒤
```

### 배열 큐 I: 인덱스 이동

`front_`는 다음 삭제 위치, `rear_`는 다음 삽입 위치입니다.

```text
[2][5][3][?][?]   front=0, rear=3
[?][5][3][?][?]   front=1, rear=3  <- 2 삭제
```

삭제한 앞쪽 공간을 다시 사용하지 못합니다. `rear_ == MAXQ`면 삽입이 막힙니다.
빈 상태는 `front_ == rear_`입니다.

### 배열 큐 II: 값 이동

항상 `arr_[0]`에서 삭제하고, 남은 값을 왼쪽으로 옮깁니다.
앞쪽 공간을 재사용할 수 있지만 삭제에 O(n)이 걸립니다.

```text
[2][5][3][?]  -- DeleteQ() -->  [5][3][?][?]
```

### 환형 큐: 끝에서 처음으로 순환

인덱스를 `(index + 1) % MAXQ`로 갱신합니다.
현재 구현은 빈 상태와 가득 찬 상태를 구분하기 위해 한 칸을 비워 둡니다.

```text
인덱스:    0   1   2   3   4
배열:     [C] [?] [?] [A] [B]
               ^       ^
              rear    front

삭제 순서: A -> B -> C
인덱스 이동: 3 -> 4 -> 0 -> 1
```

```cpp
IsEmpty: rear_ == front_
IsFull:  (rear_ + 1) % MAXQ == front_
```

배열 100칸이면 최대 저장 개수는 99개입니다.

### 포인터 큐

```text
front_ --> [2|next] --> [5|next] --> [3|0] <-- rear_
           삭제 위치                  삽입 위치
```

뒤에 노드를 연결하고 앞에서 노드를 삭제합니다.
마지막 노드를 삭제하면 front와 rear를 모두 0으로 바꿉니다.

### 응용과 확장

- `5_queueApp큐응용`: 정수 10개를 음수 큐와 나머지 큐로 나누고 각각 입력 순서로 출력합니다. 예제는 0이 없다고 가정합니다.
- `6_priorityQueue우선순위큐`: 정렬된 연결 리스트를 이용해 작은 값을 먼저 꺼냅니다. 일반 FIFO 큐와 삭제 순서가 다릅니다.
- `7_queue20퍼사드`: 용량 10인 큐 두 개를 감싸 용량 20인 큐로 사용합니다.

```text
queue20
+-------------------------------------+
| qa: 앞쪽 최대 10개 | qb: 뒤쪽 최대 10개 |
+-------------------------------------+

삭제: qa의 첫 값 제거 -> qb의 첫 값을 qa의 뒤로 이동
규칙: qb에 값이 있으면 qa는 항상 가득 찬 상태 유지
```

```cpp
#include <iostream>
using namespace std;

// [알고리즘 D.1] queue.h
#define MAXQ 100

class Queue
{
public:
    Queue();
    bool IsEmpty() { return (rear_ == front_); }
    bool IsFull() { return (rear_ == MAXQ); }
    bool InsertQ(int el);
    int DeleteQ();

private:
    int arr_[MAXQ];
    int front_;
    int rear_;
};
// 여기까지 queue.h

Queue::Queue()
{
    front_ = 0;
    rear_ = 0;
}

bool Queue::InsertQ(int el)
{
    if (rear_ == MAXQ)
        return false; // if(IsFull()) return false;
    arr_[rear_++] = el;
    return true;
}

int Queue::DeleteQ()
{
    if (rear_ == front_)
        return -1; // if(IsEmpty()) return ERROR
    return arr_[front_++];
}

int main()
{
    Queue *q = new Queue();
    q->InsertQ(2);
    q->InsertQ(5);
    q->InsertQ(3);
    while (!q->IsEmpty())
        cout << q->DeleteQ() << " ";

    delete q;
    return 0;
}
```

```cpp
{

{
    if (IsFull())
        return false;
    arr_[rear_++] = el;
    return true;

{

        return -1;
    int el = arr_[0];
    rear_--;

    return el;
}
```

```cpp
// [알고리즘 D.7] PQ의 구현
// 강의 코드에서 "구현이 있다고 가정"한 orderedList를 간단히 구현하여 포함함.

class node
{
public:
    node(int d, node *n = 0)
    {
        nxt_ = n;
        data_ = d;
    }
    // private: 코딩상의 편의를 위하여 모두 public으로 한다.
    node *nxt_;
    int data_;
};

class orderedList
{ // 오름차순으로 정렬된 연결리스트
public:
    orderedList() { lst_ = 0; }
    ~orderedList()
    {
        while (lst_ != 0)
        {
            node *ptr = lst_;
            lst_ = lst_->nxt_;
            delete ptr;
        }
    }
    orderedList(const orderedList &) = delete;
    orderedList &operator=(const orderedList &) = delete;
    int GetSize()
    {
        int n = 0;
        for (node *p = lst_; p != 0; p = p->nxt_)
            n++;
        return n;
    }
    void Insert(int item)
    {
        if (lst_ == 0 || item < lst_->data_)
        {
            lst_ = new node(item, lst_);
            return;
        }
        node *p = lst_;
        while (p->nxt_ != 0 && p->nxt_->data_ <= item)
            p = p->nxt_;
        p->nxt_ = new node(item, p->nxt_);
    }
    void Delete(int value)
    {
        if (lst_ == 0)
            return;
        node *ptr;
        if (lst_->data_ == value)
        {
            ptr = lst_;
            lst_ = lst_->nxt_;
            delete ptr;
            return;
        }
        node *p = lst_;
        while (p->nxt_ != 0 && p->nxt_->data_ != value)
            p = p->nxt_;
        if (p->nxt_ == 0)
            return;
        ptr = p->nxt_;
        p->nxt_ = ptr->nxt_;
        delete ptr;
    }
    int GetAt(int index)
    {
        node *p = lst_;
        for (int i = 0; i < index && p != 0; i++)
            p = p->nxt_;
        if (p == 0)
            return -1;
        return p->data_;
    }
    int Index(int key)
    {
        int i = 0;
        for (node *p = lst_; p != 0; p = p->nxt_, i++)
            if (p->data_ == key)
                return i;
        return -1;
    }

private:
    node *lst_;
};

class priorityQueue
{
public:
    priorityQueue()
    {
        ol_ = new orderedList();
    }
    ~priorityQueue() { delete ol_; }
    priorityQueue(const priorityQueue &) = delete;
    priorityQueue &operator=(const priorityQueue &) = delete;
    bool IsEmpty()
    {
        return ol_->GetSize() == 0;
    };
    bool IsFull()
    { // 리스트의 포인터구현에서는 full 이 없음을 가정
        return false;
    };
    bool InsertQ(int el)
    {
        ol_->Insert(el);
        return true;
    };
    int DeleteQ()
    { // list의 Delete()의 리턴타입이 int인 경우 더 간단
        int item = ol_->GetAt(0);
        ol_->Delete(item);
        return item;
    }

private:
    orderedList *ol_;
};

// [알고리즘 D.8] PQ의 응용 sorting
// priorityQueue의 구현이 priorityQueue.h에 있다고 가정함.
#include <iostream>
using namespace std;

int main() {
    priorityQueue pq;
    int v;

    for (int i=0; i<7; i++) {
        cin >> v;
        pq.InsertQ(v);
    }
    while(!pq.IsEmpty()) {
        cout << pq.DeleteQ() << " ";
    }
    cout << endl;
}
```

```cpp
// [알고리즘 D.13] queue20.h
#include <iostream>

class queue10
{
public:
    queue10() { rear = 0; }
    bool IsEmpty() { return (rear == 0); };
    bool IsFull() { return (rear == 10); };
    bool InsertQ(int el)
    {
        if (rear == 10)
            return false;
        arr[rear++] = el;
        return true;
    }
    int DeleteQ()
    {
        if (rear == 0)
            return -1;
        int el = arr[0];
        rear--;
        for (int i = 0; i < rear; i++)
            arr[i] = arr[i + 1];
        return el;
    }
    void Print()
    {
        for (int i = 0; i < rear; i++)
            std::cout << arr[i] << " ";
    }

private:
    int arr[10];
    int rear;
};

class queue20
{
public:
    queue20();
    ~queue20();
    queue20(const queue20 &) = delete;
    queue20 &operator=(const queue20 &) = delete;
    bool IsEmpty();
    bool IsFull();
    bool InsertQ(int el);
    int DeleteQ();
    void Print();
    void Reverse();

private:
    queue10 qa;
    queue10 *qb;
};

// [알고리즘 D.14] queue20.cpp
// qa: 앞쪽 10개, qb: 뒤쪽 10개
// 불변식: qb에 원소가 있으면 qa는 항상 가득 차 있다. (순서 유지)
#include <iostream>
using namespace std;

queue20::queue20()
{
    qb = new queue10();
}

queue20::~queue20()
{
    delete qb; // 생성자에서 new 한 qb 해제
}

bool queue20::IsEmpty()
{
    return qa.IsEmpty(); // qa가 비었으면 qb도 비어 있음
}

bool queue20::IsFull()
{
    return qb->IsFull(); // qb가 가득 찼으면 qa도 가득 차 있음
}

bool queue20::InsertQ(int el)
{
    if (IsFull())
        return false; // qa, qb 모두 가득 참
    if (!qa.IsFull())
        return qa.InsertQ(el);
    return qb->InsertQ(el);
}

int queue20::DeleteQ()
{
    if (IsEmpty())
        return -1;
    int el = qa.DeleteQ();
    if (!qb->IsEmpty())
        qa.InsertQ(qb->DeleteQ()); // qb의 맨 앞을 qa의 맨 뒤로 옮김
    return el;
}

void queue20::Print()
{
    cout << "qa: [ ";
    qa.Print();
    cout << "] qb: [ ";
    qb->Print();
    cout << "]" << endl;
}

void queue20::Reverse()
{
    // 전부 꺼내 임시 배열에 담은 뒤, 뒤에서부터 다시 삽입
    // InsertQ를 그대로 쓰므로 불변식(qb에 원소가 있으면 qa는 full)이 유지됨
    int buf[20], n = 0;
    while (!IsEmpty())
        buf[n++] = DeleteQ();
    for (int i = n - 1; i >= 0; i--)
        InsertQ(buf[i]);
}

int main()
{
    queue20 *q = new queue20();
    for (int i = 1; i <= 21; i++)
        if (!q->InsertQ(i))
            cout << i << " 삽입 실패(full)" << endl;
    for (int i = 0; i < 5; i++)
        cout << q->DeleteQ() << " ";
    cout << endl;
    for (int i = 22; i <= 26; i++)
        q->InsertQ(i);
    q->Print();
    q->Reverse();
    cout << "Reverse 후" << endl;
    q->Print();
    while (!q->IsEmpty())
        cout << q->DeleteQ() << " ";
    cout << endl;
    q->Print();

    delete q;
    return 0;
}
```

## 시간복잡도

n은 현재 원소 개수이며, 메모리 할당 비용은 별도입니다.

| 구현 | 삽입 | 삭제 |
| --- | --- | --- |
| 배열 I | O(1) | O(1) |
| 배열 II | O(1) | O(n) |
| 환형 / 포인터 | O(1) | O(1) |
| 현재 우선순위 큐 | O(n) | O(1) |

현재 우선순위 큐의 IsEmpty는 GetSize로 노드를 세므로 O(n)입니다.
queue20은 용량 20으로 고정되어 있습니다. 내부 큐의 삭제는 값을 이동합니다.
